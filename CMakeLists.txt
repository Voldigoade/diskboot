cmake_minimum_required(VERSION 3.25)
project(DiskBoot VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "DiskBoot requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(diskboot_core core/planner.cpp core/executor.cpp platform/windows/win.cpp platform/windows/storage.cpp platform/windows/virtual_backend.cpp boot/iso.cpp)
target_include_directories(diskboot_core PUBLIC ${CMAKE_SOURCE_DIR})
target_include_directories(diskboot_core SYSTEM PUBLIC ${CMAKE_SOURCE_DIR}/third_party)
target_link_libraries(diskboot_core PUBLIC bcrypt ole32 oleaut32 wbemuuid advapi32 shell32 virtdisk)
if(MSVC)
    target_compile_options(diskboot_core PUBLIC /W4 /WX /permissive- /EHsc /utf-8)
endif()
enable_testing()
add_executable(diskboot_tests tests/core_tests.cpp)
target_link_libraries(diskboot_tests PRIVATE diskboot_core)
add_test(NAME safety-core COMMAND diskboot_tests)
add_library(diskboot_app app/service.cpp)
target_link_libraries(diskboot_app PUBLIC diskboot_core)
add_executable(diskboot cli/main.cpp)
target_link_libraries(diskboot PRIVATE diskboot_app)
add_executable(diskboot_gui WIN32 ui/main.cpp)
target_link_libraries(diskboot_gui PRIVATE diskboot_app comdlg32 user32 gdi32)
if(MSVC)
    target_link_options(diskboot_gui PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/ui/diskboot.manifest")
endif()
add_executable(transaction_tests tests/transaction_tests.cpp)
target_link_libraries(transaction_tests PRIVATE diskboot_core)
add_test(NAME transaction-safety COMMAND transaction_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME iso-inspection COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/iso_tests.py $<TARGET_FILE:diskboot>)
add_executable(vhd_driver tests/vhd_driver.cpp)
target_link_libraries(vhd_driver PRIVATE diskboot_core)
option(DISKBOOT_VHD_TESTS "Run destructive tests on a newly created disposable VHDX only" OFF)
if(DISKBOOT_VHD_TESTS)
    add_test(NAME vhd-integration COMMAND powershell -NoProfile -File ${CMAKE_SOURCE_DIR}/tests/vhd-integration.ps1 -Driver $<TARGET_FILE:vhd_driver>)
    set_tests_properties(vhd-integration PROPERTIES SKIP_RETURN_CODE 77 LABELS virtual-storage)
endif()
